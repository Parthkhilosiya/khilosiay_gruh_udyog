${item.weight} g Rs. ${item.price.toFixed(2)} ${item.quantity} Rs. ${total.toFixed(2)} `; }); c subtotalElement.textContent = `Rs. ${subtotal.toFixed(2)}`; totalElement.textContent = `Rs. ${subtotal.toFixed(2)}`; } // Function to handle quantity change function changeQuantity(index, delta) { if (cartItems[index].quantity + delta > 0) { cartItems[index].quantity += delta; renderCart(); } } // Function to remove an item from the cart function removeItem(index) { cartItems.splice(index, 1); renderCart(); } // Clear Cart document.getElementById("clear-cart").addEventListener("click", () => { cartItems.length = 0; renderCart(); }); // Checkout (placeholder functionality) document.getElementById("checkout").addEventListener("click", () => { alert("Proceeding to checkout!"); }); // Add to Cart functionality function addToCart(event) { const productCard = event.target.closest('.product-card'); const productName = productCard.getAttribute('data-product-name'); const productPrice = parseFloat(productCard.getAttribute('data-price')); const productWeight = parseInt(productCard.getAttribute('data-weight')); // Check if item is already in the cart const existingItemIndex = cartItems.findIndex(item => item.name === productName); if (existingItemIndex > -1) { // If item exists, increase quantity cartItems[existingItemIndex].quantity += 1; } else { // Otherwise, add new item to the cart cartItems.push({ name: productName, price: productPrice, weight: productWeight, quantity: 1 }); } renderCart(); } // Attach event listeners to all "Add to Cart" buttons document.querySelectorAll('.add-to-cart-btn').forEach(button => { button.addEventListener('click', addToCart); }); // Initial cart render renderCart();
